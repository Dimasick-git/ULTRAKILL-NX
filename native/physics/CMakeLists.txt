if(NOT GIT_RESULT EQUAL 0 OR NOT JOLT_REVISION STREQUAL "01adafba08bddb650046a822983c52aa734fcadb")
 message(FATAL_ERROR "Jolt revision differs from the audited native migration dependency")

 set_target_properties(nx_physics_link PROPERTIES SUFFIX ".elf")
 target_compile_definitions(Jolt PUBLIC _POSIX_C_SOURCE=200809L)
 # libnx provides the C++ thread implementation; this bare-metal GCC target
 # does not accept the Unix -pthread driver option from upstream CMake.
 foreach(PROPERTY COMPILE_OPTIONS INTERFACE_COMPILE_OPTIONS LINK_OPTIONS INTERFACE_LINK_OPTIONS)
  get_target_property(OPTIONS Jolt ${PROPERTY})
  if(OPTIONS)
   list(REMOVE_ITEM OPTIONS -pthread)
   set_target_properties(Jolt PROPERTIES ${PROPERTY} "${OPTIONS}")
  endif()
 endforeach()
endif()
